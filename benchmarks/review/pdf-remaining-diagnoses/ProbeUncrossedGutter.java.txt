package dev.docweave.ocr;

import dev.docweave.partition.PdfOcrProvider;
import dev.docweave.partition.PdfWordOcrProvider;
import java.awt.image.BufferedImage;
import java.nio.file.Path;
import java.time.Duration;
import java.util.ArrayList;
import java.util.List;
import javax.imageio.ImageIO;

/**
 * Private diagnostic: applies the frozen UNCROSSED_GUTTER_V1 criterion to actual wrapper word output.
 * Args: tesseract tessdata then repeated "tif:dpi" sources; both pinned models run for each source.
 */
public final class ProbeUncrossedGutter {
    public static void main(final String[] args) throws Exception {
        for (int index = 2; index < args.length; index++) {
            final var parts = args[index].split(":");
            final var decoded = ImageIO.read(Path.of(parts[0]).toFile());
            // Same color-managed values the frozen harness copies into its container (getRGB).
            final var image = new BufferedImage(decoded.getWidth(), decoded.getHeight(), BufferedImage.TYPE_INT_RGB);
            final int[] row = new int[decoded.getWidth()];
            for (int y = 0; y < decoded.getHeight(); y++) {
                decoded.getRGB(0, y, decoded.getWidth(), 1, row, 0, decoded.getWidth());
                image.setRGB(0, y, decoded.getWidth(), 1, row, 0, decoded.getWidth());
            }
            for (final String language : List.of("deu", "deu_latf")) {
                final var options = new TesseractOptions(Path.of(args[0]), Path.of(args[1]), List.of(language),
                        Duration.ofSeconds(60), 1, 16_000_000, 32L * 1024 * 1024, 4 * 1024 * 1024, 10_000, 1_000_000)
                                            .withPageSegmentation(TesseractOptions.PageSegmentation.AUTO);
                final var engine =
                        TesseractOcr.createWithWords(options, TesseractThresholding.ENGINE_DEFAULT, Integer.parseInt(parts[1]));
                final var lines = engine.recognizeWords(image, PdfOcrProvider.InputKind.PAGE);
                evaluate(Path.of(parts[0]).getFileName() + " " + language, lines);
            }
        }
    }

    private static void evaluate(final String label, final List<PdfWordOcrProvider.WordLine> lines) {
        final var heights = new ArrayList<Double>();
        lines.forEach(line -> line.words().forEach(word -> heights.add(word.bottom() - word.top())));
        heights.sort(Double::compare);
        final double h = heights.get(heights.size() / 2);
        final var gaps = new ArrayList<double[]>(); // line index, left, right
        for (int line = 0; line < lines.size(); line++) {
            final var words = lines.get(line).words();
            for (int word = 1; word < words.size(); word++) {
                final double left = words.get(word - 1).right();
                final double right = words.get(word).left();
                if (right - left >= h) {
                    gaps.add(new double[] {line, left, right});
                }
            }
        }
        double best = Double.NaN;
        int bestSupport = 0;
        int candidates = 0;
        int rejectedByCrossing = 0;
        for (final var gap : gaps) {
            final double x = (gap[1] + gap[2]) / 2;
            final var support = new java.util.TreeSet<Integer>();
            for (final var other : gaps) {
                if (other[1] <= x && x <= other[2]) {
                    support.add((int) other[0]);
                }
            }
            if (support.size() < 3) {
                continue;
            }
            candidates++;
            double top = Double.POSITIVE_INFINITY;
            double bottom = Double.NEGATIVE_INFINITY;
            for (final int line : support) {
                for (final var word : lines.get(line).words()) {
                    top = Math.min(top, word.top());
                    bottom = Math.max(bottom, word.bottom());
                }
            }
            int crossing = 0;
            for (final var line : lines) {
                for (final var word : line.words()) {
                    final double center = (word.top() + word.bottom()) / 2;
                    if (word.left() < x && x < word.right() && center >= top && center <= bottom) {
                        crossing++;
                    }
                }
            }
            if (crossing > 0) {
                rejectedByCrossing++;
                continue;
            }
            if (support.size() > bestSupport || support.size() == bestSupport && x < best) {
                best = x;
                bestSupport = support.size();
            }
        }
        final String inspect = System.getProperty("inspect.x");
        if (inspect != null) {
            final double x = Double.parseDouble(inspect);
            for (final var line : lines) {
                for (final var word : line.words()) {
                    if (word.left() < x && x < word.right()) {
                        System.out.printf("  crosses %s: [%.0f,%.0f,%.0f,%.0f] len=%d%n", inspect, word.left(), word.top(),
                                word.right(), word.bottom(), word.text().length());
                    }
                }
            }
        }
        System.out.printf("%s lines=%d words=%d median_h=%.0f gaps=%d candidates>=3=%d rejected_crossing=%d "
                                  + "accepted_x=%s support=%d%n",
                label, lines.size(), heights.size(), h, gaps.size(), candidates, rejectedByCrossing,
                Double.isNaN(best) ? "none" : String.format("%.0f", best), bestSupport);
    }
}
