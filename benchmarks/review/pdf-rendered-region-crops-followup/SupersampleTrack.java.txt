import org.apache.pdfbox.Loader;import org.apache.pdfbox.rendering.*;import tools.jackson.databind.json.JsonMapper;
import java.awt.image.BufferedImage;import java.io.File;import java.nio.file.*;import java.security.MessageDigest;import java.util.*;import javax.imageio.ImageIO;
/** Diagnostic: applies the frozen 4x box supersampling policy uniformly to all 16 regions and the contract gates. */
public class SupersampleTrack{
 static final int K=4;static final double TOL=12;
 static BufferedImage sample(BufferedImage big,int x0,int y0,int w,int h){BufferedImage c=new BufferedImage(w,h,BufferedImage.TYPE_INT_RGB);
  for(int y=0;y<h;y++)for(int x=0;x<w;x++){long r=0,g=0,b=0;for(int yy=0;yy<K;yy++)for(int xx=0;xx<K;xx++){int v=big.getRGB((x0+x)*K+xx,(y0+y)*K+yy);r+=(v>>16)&255;g+=(v>>8)&255;b+=v&255;}int n=K*K;c.setRGB(x,y,(int)((r+n/2)/n)<<16|(int)((g+n/2)/n)<<8|(int)((b+n/2)/n));}return c;}
 static double mae(BufferedImage a,BufferedImage b){double t=0;for(int y=0;y<a.getHeight();y++)for(int x=0;x<a.getWidth();x++){int p=a.getRGB(x,y),q=b.getRGB(x,y);for(int s=0;s<24;s+=8)t+=Math.abs(((p>>s)&255)-((q>>s)&255));}return t/(3.0*a.getWidth()*a.getHeight());}
 static List<Integer> rgb(int p){return List.of((p>>16)&255,(p>>8)&255,p&255);}
 static String sha(Path p)throws Exception{return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(Files.readAllBytes(p)));}
 public static void main(String[] a)throws Exception{
  var m=JsonMapper.builder().build();var out=new LinkedHashMap<String,Object>();out.put("policy_sha256",sha(Path.of("/tmp/docweave-rendered-crops-followup21/supersample-policy.json")));
  var contract=new ArrayList<Object>();boolean gates=true;
  try(var doc=Loader.loadPDF(new File("/tmp/docweave-gap-comparison21/images/image-contract.pdf"))){for(int pg=0;pg<3;pg++){var big=new PDFRenderer(doc).renderImage(pg,1f*K,ImageType.RGB);var c=sample(big,40,40,40,40);var l=rgb(c.getRGB(5,20));var r=rgb(c.getRGB(35,20));
   boolean pass=switch(pg){case 0->l.stream().allMatch(v->v>=250)&&r.stream().allMatch(v->v>=250);case 1->l.get(0)>=250&&l.get(1)>=120&&l.get(1)<=135&&l.get(2)>=120&&l.get(2)<=135;default->l.get(0)>=245&&l.get(1)<=10&&l.get(2)<=10&&r.stream().allMatch(v->v>=250);};
   gates&=pass;contract.add(Map.of("page",pg+1,"left",l,"right",r,"gate",pass));}}
  out.put("contract",contract);out.put("contract_gates_pass",gates);
  var transfer=m.readTree(Files.readAllBytes(Path.of("/tmp/docweave-rendered-crops-followup21/transfer-java21.json")));
  var capture=m.readTree(Files.readAllBytes(Path.of("/tmp/docweave-crop-review-20261022/upstream-capture.json")));
  var jpeg=new HashMap<Integer,String>();for(var c:capture.get("crops"))jpeg.put(c.get("occurrence_index").asInt(),c.get("filename").asString());
  var rows=new ArrayList<Object>();int[] base=new int[3],cand=new int[3];
  try(var doc=Loader.loadPDF(new File("/tmp/docweave-caption-negatives21/immunology-news-march-2025.pdf"))){
   var pages=new TreeMap<Integer,List<tools.jackson.databind.JsonNode>>();for(var r:transfer.get("magazine").get("regions"))pages.computeIfAbsent(r.get("page").asInt(),k->new ArrayList<>()).add(r);
   for(var e:pages.entrySet()){var big=new PDFRenderer(doc).renderImage(e.getKey()-1,(float)(150/72.0)*K,ImageType.RGB);
    for(var r:e.getValue()){int i=r.get("occurrence_index").asInt();var rect=r.get("pixel_rect");
     var c=sample(big,rect.get(0).asInt(),rect.get(1).asInt(),rect.get(2).asInt(),rect.get(3).asInt());
     var own=ImageIO.read(new File("/tmp/docweave-rendered-crops-followup21/crops/docweave-"+i+".png"));
     var refs=new BufferedImage[]{ImageIO.read(new File("/tmp/docweave-rendered-crops-followup21/crops/poppler-"+i+".png")),ImageIO.read(new File("/tmp/docweave-crop-review-20261022/upstream-lossless/crop-"+i+".png")),ImageIO.read(new File("/tmp/docweave-crop-review-20261022/upstream-crops/"+jpeg.get(i)))};
     var row=new LinkedHashMap<String,Object>();row.put("occurrence_index",i);String[] names={"poppler","pdfium_lossless","unstructured_jpeg"};
     for(int t=0;t<3;t++){double b=mae(own,refs[t]),s=mae(c,refs[t]);if(b<=TOL)base[t]++;if(s<=TOL)cand[t]++;row.put(names[t],List.of(Math.round(b*1000)/1000.0,Math.round(s*1000)/1000.0));}
     rows.add(row);}
    big.flush();}}
  rows.sort(Comparator.comparingInt(o->(Integer)((Map<?,?>)o).get("occurrence_index")));
  out.put("columns","[current 1x, frozen 4x supersampled] MAE per track");
  out.put("within_tolerance",Map.of("poppler",List.of(base[0],cand[0]),"pdfium_lossless",List.of(base[1],cand[1]),"unstructured_jpeg",List.of(base[2],cand[2])));
  out.put("regions",rows);
  Files.writeString(Path.of(a[0]),m.writerWithDefaultPrettyPrinter().writeValueAsString(out)+"\n",StandardOpenOption.CREATE_NEW);
  System.out.println(out.get("contract_gates_pass")+" "+out.get("within_tolerance"));}}
