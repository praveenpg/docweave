import dev.docweave.partition.*;
import java.awt.Color;
import java.io.*;
import java.util.List;
import javax.imageio.ImageIO;
import org.apache.pdfbox.pdmodel.*;
import org.apache.pdfbox.pdmodel.common.PDRectangle;
/** P8: hand-derived edge-pixel expectations for BOX_4X under both backgrounds. */
public class ProbeEdgeAlpha {
    public static void main(String[] a) throws Exception {
        byte[] src; try (var d = new PDDocument()) { var p = new PDPage(new PDRectangle(200,120)); d.addPage(p);
            try (var c = new PDPageContentStream(d,p)) { c.setNonStrokingColor(Color.RED); c.addRect(40, 40, 20.25f, 40); c.fill();
                // vertical edge at x=100.5: half coverage; horizontal edge at y(top-frame)=... use rect from x=100 to 120.5
                c.setNonStrokingColor(new Color(0,0,255)); c.addRect(100, 40, 20.5f, 40); c.fill(); }
            var o = new ByteArrayOutputStream(); d.save(o); src = o.toByteArray(); }
        for (var bg : PdfPageCropOptions.Background.values()) for (var s : PdfPageCropOptions.Sampling.values()) {
            var crops = PdfPageCropRenderer.create(PdfPageCropOptions.defaults().withDpi(72).withBackground(bg).withSampling(s))
                .render(DocumentSource.bytes(src, null), List.of(new PdfPageCropRequest(1, 40, 40, 80, 80), new PdfPageCropRequest(1, 100, 40, 140, 80)));
            var r = ImageIO.read(new ByteArrayInputStream(crops.get(0).png())); var b = ImageIO.read(new ByteArrayInputStream(crops.get(1).png()));
            System.out.printf("%s %s quarter-edge(20,20)=%08x interior(10,20)=%08x outside(25,20)=%08x | half-edge(20,20)=%08x interior(10,20)=%08x%n", bg, s,
                r.getRGB(20,20), r.getRGB(10,20), r.getRGB(25,20), b.getRGB(20,20), b.getRGB(10,20));
        }
    }
}
