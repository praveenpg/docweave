import java.awt.color.*;
import java.nio.file.*;
import javax.imageio.ImageIO;
public class ProbeBlack { public static void main(String[] a) throws Exception {
  try (var in = ImageIO.createImageInputStream(Path.of(a[0]).toFile())) { var r = ImageIO.getImageReaders(in).next(); r.setInput(in);
    var cs = r.read(0).getColorModel().getColorSpace(); r.dispose();
    for (float v : new float[]{0f, 0.1f, 0.25f, 0.5f, 1f}) { float[] o = cs.toRGB(new float[]{v,v,v}); System.out.printf("stored %.2f -> sRGB %.4f %.4f %.4f (8bit %d)%n", v, o[0], o[1], o[2], Math.round(o[1]*255)); } } } }
