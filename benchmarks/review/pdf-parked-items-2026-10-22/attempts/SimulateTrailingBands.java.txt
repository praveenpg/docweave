import java.awt.geom.*;
import java.io.File;
import java.nio.file.*;
import java.util.*;
import org.apache.pdfbox.Loader;
import org.apache.pdfbox.contentstream.PDFGraphicsStreamEngine;
import org.apache.pdfbox.cos.COSName;
import org.apache.pdfbox.pdmodel.PDPage;
import org.apache.pdfbox.pdmodel.graphics.image.PDImage;
import org.apache.pdfbox.text.*;
import tools.jackson.databind.json.JsonMapper;
/** P3 frozen simulations (band lattice; numeric continuation). Args: capture.json pdf mode(BANDS|NUMERIC) */
public class SimulateTrailingBands {
    record Rect(double l, double t, double r, double b) {}
    record Glyph(String u, double x, double w, double base) {}
    record Line(double base, List<Glyph> glyphs) { String text() { var s = new StringBuilder(); Glyph prev = null; for (var g : glyphs) { if (prev != null && g.x() - (prev.x() + prev.w()) > 1.5) s.append(' '); s.append(g.u()); prev = g; } return s.toString().replaceAll("\\s+", " ").trim(); } }
    public static void main(String[] a) throws Exception {
        var pages = JsonMapper.builder().build().readTree(Files.readAllBytes(Path.of(a[0])));
        final double k = 72.0 / 150;
        int appendedTotal = 0;
        try (var doc = Loader.loadPDF(new File(a[1]))) {
            for (int p = 0; p < pages.size(); p++) {
                if (!pages.get(p).has("tables") || pages.get(p).get("tables").isEmpty()) continue;
                var page = doc.getPage(p); double h = page.getCropBox().getHeight();
                var rects = new ArrayList<Rect>();
                new PDFGraphicsStreamEngine(page) {
                    GeneralPath path = new GeneralPath();
                    void add() { var bb = path.getBounds2D(); if (bb.getWidth() > 0 && bb.getHeight() > 0) rects.add(new Rect(bb.getMinX(), h - bb.getMaxY(), bb.getMaxX(), h - bb.getMinY())); path.reset(); }
                    @Override public void appendRectangle(Point2D p0, Point2D p1, Point2D p2, Point2D p3) { path.moveTo(p0.getX(), p0.getY()); path.lineTo(p1.getX(), p1.getY()); path.lineTo(p2.getX(), p2.getY()); path.lineTo(p3.getX(), p3.getY()); path.closePath(); }
                    @Override public void drawImage(PDImage i) {}
                    @Override public void clip(int r) { path.reset(); }
                    @Override public void moveTo(float x, float y) { path.moveTo(x, y); }
                    @Override public void lineTo(float x, float y) { path.lineTo(x, y); }
                    @Override public void curveTo(float x1, float y1, float x2, float y2, float x3, float y3) { path.curveTo(x1, y1, x2, y2, x3, y3); }
                    @Override public Point2D getCurrentPoint() { return path.getCurrentPoint(); }
                    @Override public void closePath() { path.closePath(); }
                    @Override public void endPath() { path.reset(); }
                    @Override public void strokePath() { path.reset(); }
                    @Override public void fillPath(int r) { add(); }
                    @Override public void fillAndStrokePath(int r) { add(); }
                    @Override public void shadingFill(COSName n) {}
                }.processPage(page);
                var glyphs = new ArrayList<Glyph>(); final int pn = p + 1;
                var stripper = new PDFTextStripper() { @Override protected void processTextPosition(TextPosition t) { if (!t.getUnicode().isBlank()) glyphs.add(new Glyph(t.getUnicode(), t.getXDirAdj(), t.getWidthDirAdj(), t.getYDirAdj())); } };
                stripper.setStartPage(pn); stripper.setEndPage(pn); stripper.getText(doc);
                glyphs.sort(Comparator.comparingDouble(Glyph::base).thenComparingDouble(Glyph::x));
                var lines = new ArrayList<Line>();
                for (var g : glyphs) { if (!lines.isEmpty() && Math.abs(lines.getLast().base() - g.base()) <= 2.0) lines.getLast().glyphs().add(g); else lines.add(new Line(g.base(), new ArrayList<>(List.of(g)))); }
                for (var line : lines) line.glyphs().sort(Comparator.comparingDouble(Glyph::x));
                int ti = 0;
                for (var t : pages.get(p).get("tables")) {
                    ti++;
                    var bb = t.get("bounds"); double tl = bb.get("left").asDouble() * k, tt = bb.get("top").asDouble() * k, tr = bb.get("right").asDouble() * k, tb = bb.get("bottom").asDouble() * k;
                    int rows = t.get("rows").asInt();
                    var rowTop = new double[rows]; var rowBottom = new double[rows]; Arrays.fill(rowTop, 1e9);
                    var last = new ArrayList<double[]>();
                    for (var x : t.get("anchors")) { int r = x.get("row").asInt(); var b = x.get("bounds"); rowTop[r] = Math.min(rowTop[r], b.get("top").asDouble() * k); rowBottom[r] = Math.max(rowBottom[r], b.get("bottom").asDouble() * k);
                        if (r + x.get("rowSpan").asInt() == rows) last.add(new double[]{b.get("left").asDouble() * k, b.get("right").asDouble() * k, x.get("column").asInt()}); }
                    last.sort(Comparator.comparingDouble(c -> c[0]));
                    var inTable = lines.stream().filter(l -> l.base() >= tt && l.base() < tb && l.glyphs().stream().anyMatch(g -> g.x() >= tl && g.x() < tr)).toList();
                    var lastLine = inTable.isEmpty() ? null : inTable.getLast();
                    var below = lines.stream().filter(l -> l.base() >= tb && l.base() <= tb + 120 && l.glyphs().stream().allMatch(g -> g.x() + g.w() / 2 >= tl - 2 && g.x() + g.w() / 2 <= tr + 2)).toList();
                    var appended = a[2].equals("BANDS") ? bands(rects, inTable, below, tl, tr, last) : numeric(inTable, below, last, rowTop, rowBottom, rows);
                    appendedTotal += appended.size();
                    System.out.printf("page %d table %d rows=%d bottom=%.1f appended=%d%n", pn, ti, rows, tb, appended.size());
                    for (var l : appended) System.out.printf("   + y=%.1f | %s%n", l.base(), l.text());
                }
            }
        }
        System.out.println("appended_total=" + appendedTotal);
    }
    static boolean fits(Line l, List<double[]> cols) { return l.glyphs().stream().allMatch(g -> { double c = g.x() + g.w() / 2; return cols.stream().anyMatch(r -> c >= r[0] && c < r[1]); }); }
    static List<Line> bands(List<Rect> rects, List<Line> inTable, List<Line> below, double tl, double tr, List<double[]> last) {
        var groups = new TreeMap<String, double[]>();
        for (var r : rects) { if (r.b() - r.t() < 4) continue; String key = Math.round(r.t() * 2) + ":" + Math.round(r.b() * 2); groups.computeIfAbsent(key, x -> new double[]{r.t(), r.b(), 0}); }
        var bands = new ArrayList<double[]>();
        for (var e : groups.entrySet()) { double t = e.getValue()[0], b = e.getValue()[1]; var spans = rects.stream().filter(r -> Math.abs(r.t() - t) <= 0.5 && Math.abs(r.b() - b) <= 0.5).map(r -> new double[]{Math.max(r.l(), tl), Math.min(r.r(), tr)}).filter(s -> s[1] > s[0]).sorted(Comparator.comparingDouble(s -> s[0])).toList();
            double cover = 0, end = tl; for (var s : spans) { if (s[1] > end) { cover += s[1] - Math.max(s[0], end); end = s[1]; } }
            if (cover >= 0.9 * (tr - tl)) bands.add(new double[]{t, b}); }
        bands.sort(Comparator.comparingDouble(x -> x[0]));
        long covered = inTable.stream().filter(l -> bands.stream().anyMatch(b -> l.base() >= b[0] && l.base() <= b[1])).count();
        if (inTable.size() < 2 || covered < inTable.size()) return List.of();
        var lastLine = inTable.getLast();
        double[] current = bands.stream().filter(b -> lastLine.base() >= b[0] && lastLine.base() <= b[1]).findFirst().orElse(null);
        var out = new ArrayList<Line>();
        while (current != null) {
            final double[] prev = current;
            double[] next = bands.stream().filter(b -> b[0] > prev[0] && Math.abs(b[0] - prev[1]) <= 1.5).findFirst().orElse(null);
            if (next == null) break;
            var text = below.stream().filter(l -> l.base() >= next[0] && l.base() <= next[1]).toList();
            if (!text.isEmpty()) { if (!text.stream().allMatch(l -> fits(l, last))) break; out.addAll(text); }
            current = next;
        }
        return out;
    }
    static List<Line> numeric(List<Line> inTable, List<Line> below, List<double[]> last, double[] rowTop, double[] rowBottom, int rows) {
        if (inTable.size() < 2 || last.isEmpty()) return List.of();
        var bases = inTable.stream().mapToDouble(Line::base).toArray(); var pitches = new double[bases.length - 1]; for (int i = 1; i < bases.length; i++) pitches[i - 1] = bases[i] - bases[i - 1]; Arrays.sort(pitches); double pitch = pitches[pitches.length / 2];
        var lastLine = inTable.getLast();
        var valueCols = last.stream().skip(1).filter(c -> lastLine.glyphs().stream().anyMatch(g -> g.x() + g.w() / 2 >= c[0] && g.x() + g.w() / 2 < c[1])).toList();
        if (System.getenv("P3DEBUG") != null) { System.out.println("  dbg last=" + lastLine.base() + " " + lastLine.text() + " pitch=" + pitch + " valueCols=" + valueCols.size()); for (var l : below) System.out.println("  dbg below " + l.base() + " tokens=" + tokens(l).stream().map(g -> g.u() + "@" + Math.round(g.x())).toList()); }
        var out = new ArrayList<Line>(); double prevBase = lastLine.base(); Line pending = null;
        for (var l : below) {
            if (l.base() - prevBase > 1.5 * pitch * (pending == null ? 1 : 2)) break;
            var tokens = tokens(l); if (tokens.isEmpty()) break;
            var first = tokens.getFirst(); double[] c0 = last.getFirst();
            if (!(first.x() >= c0[0] && first.x() < c0[1])) break;
            var rest = tokens.subList(1, tokens.size());
            if (rest.isEmpty() && l.text().endsWith(":") && pending == null) { pending = l; continue; }
            boolean ok = !valueCols.isEmpty() && rest.size() == valueCols.size();
            for (int i = 0; ok && i < rest.size(); i++) { var tk = rest.get(i); double c = tk.x() + tk.w() / 2; ok = tk.u().matches("[$0-9,.()%]+") && c >= valueCols.get(i)[0] && c < valueCols.get(i)[1]; }
            if (!ok) break;
            if (pending != null) { out.add(pending); pending = null; }
            out.add(l); prevBase = l.base();
        }
        return out;
    }
    static List<Glyph> tokens(Line l) { var out = new ArrayList<Glyph>(); Glyph start = null; var sb = new StringBuilder(); Glyph prev = null; double end = 0;
        for (var g : l.glyphs()) { if (prev != null && g.x() - (prev.x() + prev.w()) > 4) { out.add(new Glyph(sb.toString(), start.x(), end - start.x(), l.base())); sb.setLength(0); start = null; } if (start == null) start = g; sb.append(g.u()); end = g.x() + g.w(); prev = g; }
        if (start != null) out.add(new Glyph(sb.toString(), start.x(), end - start.x(), l.base())); return out; }
}
