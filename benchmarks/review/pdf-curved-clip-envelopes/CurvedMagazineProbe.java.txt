package dev.docweave.validation;

import dev.docweave.DocWeave;
import dev.docweave.core.Element;
import dev.docweave.core.ElementType;
import dev.docweave.partition.*;
import java.nio.file.*;
import java.util.*;

/** Candidate check: all occurrences, envelopes with/without curved-clip envelopes; additive-only invariants. */
public final class CurvedMagazineProbe {
    public static void main(String[] a) throws Exception {
        var bytes = Files.readAllBytes(Path.of(a[0]));
        var base = PdfOptions.defaults().withTextLayout(PdfOptions.TextLayout.ORIENTED_LINES).withImageClipBounds(true);
        var before = images(bytes, base);
        var after = images(bytes, base.withCurvedClipEnvelopes(true));
        if (before.size() != after.size()) throw new IllegalStateException("population changed");
        int tighter = 0, same = 0, newlyExcluded = 0, widened = 0; double areaBefore = 0, areaAfter = 0;
        var perPage = new TreeMap<Object, Integer>();
        for (int i = 0; i < before.size(); i++) {
            var m0 = new LinkedHashMap<>(before.get(i).metadata()); var m1 = new LinkedHashMap<>(after.get(i).metadata());
            var c0 = (Map<?, ?>) m0.remove("pdf_image_clip_bounds"); var c1 = (Map<?, ?>) m1.remove("pdf_image_clip_bounds");
            if (!m0.equals(m1)) throw new IllegalStateException("other metadata changed at " + i);
            double[] b0 = box(c0), b1 = box(c1);
            if (b1[0] < b0[0] - 1e-6 || b1[1] < b0[1] - 1e-6 || b1[2] > b0[2] + 1e-6 || b1[3] > b0[3] + 1e-6) widened++;
            if (Arrays.equals(b0, b1)) same++; else { tighter++; perPage.merge(m0.get("page_number"), 1, Integer::sum); }
            areaBefore += (b0[2] - b0[0]) * (b0[3] - b0[1]); areaAfter += (b1[2] - b1[0]) * (b1[3] - b1[1]);
            if (!Boolean.TRUE.equals(c0.get("excludes_image_placement")) && Boolean.TRUE.equals(c1.get("excludes_image_placement"))) newlyExcluded++;
            if (!Arrays.equals(b0, b1)) System.out.printf("occurrence %d page %s placement=%s before=%s after=%s excluded=%s%n", i, m0.get("page_number"),
                    ((Map<?, ?>) m0.get("coordinates")).get("points"), Arrays.toString(round(b0)), Arrays.toString(round(b1)), c1.get("excludes_image_placement"));
        }
        System.out.printf("occurrences=%d unchanged=%d tighter=%d widened=%d newly_excluded=%d tighter_by_page=%s total_envelope_area %.0f -> %.0f%n",
                before.size(), same, tighter, widened, newlyExcluded, perPage, areaBefore, areaAfter);
    }
    static double[] round(double[] b) { return Arrays.stream(b).map(v -> Math.round(v * 10) / 10.0).toArray(); }
    static double[] box(Map<?, ?> c) {
        if (!"BOUNDED".equals(c.get("classification"))) return new double[] {0, 0, 0, 0};
        var p = (List<?>) ((Map<?, ?>) c.get("bounds")).get("points");
        return new double[] {n(p, 0, 0), n(p, 0, 1), n(p, 2, 0), n(p, 2, 1)};
    }
    static double n(List<?> p, int i, int j) { return ((Number) ((List<?>) p.get(i)).get(j)).doubleValue(); }
    static List<Element> images(byte[] bytes, PdfOptions o) throws Exception {
        return DocWeave.create(PartitionerRegistry.of(List.of(PdfPartitioner.create(o)))).partition(DocumentSource.bytes(bytes, "m.pdf"))
                .elements().stream().filter(e -> e.type() == ElementType.Image).toList();
    }
}
