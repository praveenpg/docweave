package dev.docweave.ocr;

import dev.docweave.partition.PdfOcrProvider;
import dev.docweave.partition.PdfWordOcrProvider;
import java.awt.image.BufferedImage;
import java.nio.file.Path;
import java.time.Duration;
import java.util.ArrayList;
import java.util.List;
import javax.imageio.ImageIO;

/**
 * Private diagnostic for PRINTED_RULE_DIAGNOSTIC_V2_LOCAL_INK. Args: tesseract tessdata then repeated "tif:dpi" sources; both
 * pinned models run for each source.
 */
public final class ProbeLocalInkRule {
    public static void main(final String[] args) throws Exception {
        for (int index = 2; index < args.length; index++) {
            final var parts = args[index].split(":");
            final var decoded = ImageIO.read(Path.of(parts[0]).toFile());
            final var image = new BufferedImage(decoded.getWidth(), decoded.getHeight(), BufferedImage.TYPE_INT_RGB);
            final int[] row = new int[decoded.getWidth()];
            for (int y = 0; y < decoded.getHeight(); y++) {
                decoded.getRGB(0, y, decoded.getWidth(), 1, row, 0, decoded.getWidth());
                image.setRGB(0, y, decoded.getWidth(), 1, row, 0, decoded.getWidth());
            }
            for (final String language : List.of("deu", "deu_latf")) {
                final var options = new TesseractOptions(Path.of(args[0]), Path.of(args[1]), List.of(language),
                        Duration.ofSeconds(60), 1, 16_000_000, 32L * 1024 * 1024, 4 * 1024 * 1024, 10_000, 1_000_000)
                                            .withPageSegmentation(TesseractOptions.PageSegmentation.AUTO);
                final var lines = TesseractOcr.createWithWords(options, TesseractThresholding.ENGINE_DEFAULT,
                                                      Integer.parseInt(parts[1]))
                                          .recognizeWords(image, PdfOcrProvider.InputKind.PAGE);
                final var heights = new ArrayList<Double>();
                lines.forEach(line -> line.words().forEach(word -> heights.add(word.bottom() - word.top())));
                heights.sort(Double::compare);
                final double h = heights.get(heights.size() / 2);
                double left = Double.MAX_VALUE, top = Double.MAX_VALUE, right = -Double.MAX_VALUE, bottom = -Double.MAX_VALUE;
                for (final var line : lines) {
                    for (final var word : line.words()) {
                        left = Math.min(left, word.left());
                        top = Math.min(top, word.top());
                        right = Math.max(right, word.right());
                        bottom = Math.max(bottom, word.bottom());
                    }
                }
                final int[] area = {(int) Math.max(0, Math.floor(left - h)), (int) Math.max(0, Math.floor(top - h)),
                        (int) Math.min(image.getWidth() - 1, Math.ceil(right + h)),
                        (int) Math.min(image.getHeight() - 1, Math.ceil(bottom + h))};
                System.out.printf("text_area=%d,%d-%d,%d%n", area[0], area[1], area[2], area[3]);
                evaluate(Path.of(parts[0]).getFileName() + " " + language, ink(image, area), lines);
            }
        }
    }

    private static boolean[][] ink(final BufferedImage image, final int[] area) {
        final int width = image.getWidth();
        final int height = image.getHeight();
        final int[] histogram = new int[256];
        final int[][] luminance = new int[height][width];
        for (int y = 0; y < height; y++) {
            for (int x = 0; x < width; x++) {
                final int rgb = image.getRGB(x, y);
                final int value = (int) Math.round(
                        0.299 * ((rgb >> 16) & 255) + 0.587 * ((rgb >> 8) & 255) + 0.114 * (rgb & 255));
                luminance[y][x] = value;
                if (x >= area[0] && x <= area[2] && y >= area[1] && y <= area[3]) {
                    histogram[value]++;
                }
            }
        }
        // Otsu threshold.
        final long total = (long) (area[2] - area[0] + 1) * (area[3] - area[1] + 1);
        double sum = 0;
        for (int value = 0; value < 256; value++) {
            sum += value * (double) histogram[value];
        }
        double sumBackground = 0;
        long weightBackground = 0;
        double best = -1;
        int threshold = 0;
        for (int value = 0; value < 256; value++) {
            weightBackground += histogram[value];
            if (weightBackground == 0) {
                continue;
            }
            final long weightForeground = total - weightBackground;
            if (weightForeground == 0) {
                break;
            }
            sumBackground += value * (double) histogram[value];
            final double meanBackground = sumBackground / weightBackground;
            final double meanForeground = (sum - sumBackground) / weightForeground;
            final double between = (double) weightBackground * weightForeground
                                   * (meanBackground - meanForeground) * (meanBackground - meanForeground);
            if (between > best) {
                best = between;
                threshold = value;
            }
        }
        final boolean[][] ink = new boolean[height][width];
        for (int y = 0; y < height; y++) {
            for (int x = 0; x < width; x++) {
                ink[y][x] = x >= area[0] && x <= area[2] && y >= area[1] && y <= area[3]
                            && luminance[y][x] <= threshold;
            }
        }
        System.out.println("otsu_threshold=" + threshold);
        return ink;
    }

    private static void evaluate(final String label, final boolean[][] ink,
            final List<PdfWordOcrProvider.WordLine> lines) {
        final var heights = new ArrayList<Double>();
        lines.forEach(line -> line.words().forEach(word -> heights.add(word.bottom() - word.top())));
        heights.sort(Double::compare);
        final double h = heights.get(heights.size() / 2);
        final int height = ink.length;
        final int width = ink[0].length;
        final int[] longest = new int[width];
        final int[] start = new int[width];
        for (int x = 2; x < width - 2; x++) {
            int run = 0;
            for (int y = 0; y < height; y++) {
                final boolean any = ink[y][x - 2] || ink[y][x - 1] || ink[y][x] || ink[y][x + 1] || ink[y][x + 2];
                run = any ? run + 1 : 0;
                if (run > longest[x]) {
                    longest[x] = run;
                    start[x] = y - run + 1;
                }
            }
        }
        final int minimum = (int) Math.ceil(8 * h);
        System.out.printf("%s median_h=%.0f min_run=%d%n", label, h, minimum);
        for (int x = 0; x < width; x++) {
            if (longest[x] < minimum) {
                continue;
            }
            int end = x;
            int bestX = x;
            while (end + 1 < width && longest[end + 1] >= minimum) {
                end++;
                if (longest[end] > longest[bestX]) {
                    bestX = end;
                }
            }
            final double center = (x + end) / 2.0;
            final int y0 = start[bestX];
            final int y1 = y0 + longest[bestX] - 1;
            int support = 0;
            int ruleTokens = 0;
            int crossing = 0;
            for (final var line : lines) {
                final double middle = (line.line().top() + line.line().bottom()) / 2;
                if (middle < y0 || middle > y1) {
                    continue;
                }
                boolean leftWord = false;
                boolean rightWord = false;
                for (final var word : line.words()) {
                    if (word.right() <= center - 2) {
                        leftWord = true;
                    } else if (word.left() >= center + 2) {
                        rightWord = true;
                    } else if (word.text().codePointCount(0, word.text().length()) == 1) {
                        ruleTokens++;
                    } else {
                        crossing++;
                    }
                }
                support += leftWord && rightWord ? 1 : 0;
            }
            final String verdict = crossing > 0 ? "rejected_crossing_word"
                                   : support < 3 ? "rejected_support" : "accepted";
            System.out.printf("  candidate x=%d-%d center=%.1f extent=%d-%d run=%d support=%d rule_tokens=%d "
                                      + "crossing_words=%d %s%n",
                    x, end, center, y0, y1, longest[bestX], support, ruleTokens, crossing, verdict);
            x = end;
        }
    }
}
